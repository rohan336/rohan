Q1.#include <stdio.h>
#include <unistd.h>
#include <fcntl.h>
int main() {
 int fd;
 // Open output.txt for writing (create if not exists)
 fd = open("output.txt", O_WRONLY | O_CREAT | O_TRUNC, 0644);
 if (fd < 0) {
 perror("open");
 return 1;
 }
 // Redirect standard output (fd = 1) to file using dup2
 dup2(fd, STDOUT_FILENO);
 // Now printf will go to output.txt instead of terminal
 printf("This line will be written into output.txt!\n");
 printf("Standard output successfully redirected.\n");
 close(fd);
 return 0;
}
Q2.#include <stdio.h>
#include <unistd.h>
#include <stdlib.h>
int main() {
int fd[2];
 // Create pipe
 if (pipe(fd) == -1) {
 perror("pipe");
 exit(1);
 }
 // First child → executes "ls -l"
 if (fork() == 0) {
 dup2(fd[1], STDOUT_FILENO); // write end to stdout
 close(fd[0]); // close read end
 close(fd[1]);
 execlp("ls", "ls", "-l", NULL);
 perror("execlp ls");
 exit(1);
 }
 // Second child → executes "wc -l"
 if (fork() == 0) {
 dup2(fd[0], STDIN_FILENO); // read end to stdin
 close(fd[1]); // close write end
 close(fd[0]);
 execlp("wc", "wc", "-l", NULL);
 perror("execlp wc");
exit(1);
 }
 // Parent process closes pipe ends
 close(fd[0]);
 close(fd[1]);
 // Wait for both children
 wait(NULL);
 wait(NULL);
 return 0;
}
