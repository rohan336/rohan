Q1.#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int main(int argc, char *argv[])
{
    char cmd[50];

    if (argc == 2)
    {
        memset(cmd, '\0', sizeof(cmd));   // Clear buffer

        strcpy(cmd, "ls ");               // base command
        strcat(cmd, "> ");                // redirection operator
        strcat(cmd, argv[1]);             // output filename

        system(cmd);                      // execute the command
    }
    else
    {
        printf("Invalid number of inputs\n");
    }

    return 0;
}
Q2.#include <stdio.h>
#include <fcntl.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/wait.h>
int main()
{
    int pid;
    char file[30];

    pid = fork();

    if (pid < 0)
    {
        printf("Fork failed\n");
        exit(-1);
    }
    else if (pid == 0)
    {
        printf("Enter the file name: ");
        scanf("%s", file);

        // Execute ls -l <file>
        execlp("/bin/ls", "ls", "-l", file, NULL);

        printf("Exec failed\n");   // runs only if exec fails
        exit(1);
    }
    else
    {
        wait(NULL);
        printf("Child completed\n");
        exit(0);
    }

    return 0;
}
