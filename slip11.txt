Q1.#include <stdio.h>
#include <stdlib.h>
#include <sys/resource.h>
int main() {
 struct rlimit limit;
 // Get current file limit
 getrlimit(RLIMIT_NOFILE, &limit);
 printf("Current File Limit: soft = %ld, hard = %ld\n",
 limit.rlim_cur, limit.rlim_max);
 // Set new file soft limit
 limit.rlim_cur = 1024; // soft limit
 limit.rlim_max = 2048; // hard limit
 setrlimit(RLIMIT_NOFILE, &limit);
 // Get updated file limit
 getrlimit(RLIMIT_NOFILE, &limit);
 printf("Updated File Limit: soft = %ld, hard = %ld\n",
 limit.rlim_cur, limit.rlim_max);
 // Get memory limit
 getrlimit(RLIMIT_AS, &limit);
 printf("Current Memory Limit (bytes): soft = %ld, hard = %ld\n",
 limit.rlim_cur, limit.rlim_max
return 0;
}
Q2.#include <stdlib.h>
#include <unistd.h>
#include <fcntl.h>
int main() {
 int fd;
 // Open the file for writing (create if not exists)
 fd = open("output.txt", O_WRONLY | O_CREAT | O_TRUNC, 0644);
 if (fd < 0) {
 perror("open");
 exit(1);
 }
 // Redirect stdout (file descriptor 1) to output.txt
 dup2(fd, STDOUT_FILENO);
 // Now all printf statements will go to output.txt
 printf("This output goes to output.txt file!\n");
 close(fd);
 return 0;
}
