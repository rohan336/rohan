Q1.#include<stdio.h>
#include<sys/stat.h>
#include<stdlib.h>
int main(int argc , char*argv[])
{
struct stat * st;
st=(struct stat*)malloc (sizeof (struct stat));
stat(argv[1],st);
if(S_ISDIR(st->st_mode))
printf("file type is :Directory\n");
else if(S_ISCHR(st->st_mode))
printf("file type is :character device\n");
else if(S_ISBLK(st->st_mode))
printf("file type is :Block device\n");
else if(S_ISFIFO(st->st_mode))
printf("file type is fifo or pipe\n");
return 0;
}
Q2.#include <signal.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/wait.h>
#include <unistd.h>
pid_t child_pid;
// Handler for SIGCHLD
void sigchld_handler(int sig)
{
 printf("Received SIGCHLD signal\n");
}
// Handler for SIGALRM
void sigalrm_handler(int sig)
{
 printf("Received SIGALRM signal. Killing child process...\n");
 kill(child_pid, SIGKILL);
}
int main()
{
 // Set signal handlers
 signal(SIGCHLD, sigchld_handler);
 signal(SIGALRM, sigalrm_handler);
 child_pid = fork();
 if (child_pid < 0)
 {
 perror("fork");
 exit(1);
 }
 else if (child_pid == 0)
 {
 // Child process runs a command
execlp("/bin/ls", "ls", NULL);
 perror("exec");
 exit(1);
 }
 else
 {
 alarm(5); // Child must finish within 5 sec
 int status;
 wait(&status);
 alarm(0); // Cancel alarm when child is done
 if (WIFEXITED(status))
 {
 printf("Child exited normally with status %d\n",
 WEXITSTATUS(status));
 }
 else if (WIFSIGNALED(status))
 {
 printf("Child was killed by signal %d\n",
 WTERMSIG(status));
 }
 }
 return 0;
}
