Q1.#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/stat.h>
int main(int argc, char *argv[])
{
 int i;
 struct stat buf;
 ino_t inode;
 if (argc < 2)
 {
 printf("Usage: %s <filename>\n", argv[0]);
 return 1;
 }
 for (i = 1; i < argc; i++)
 {
 printf("File is: %s\n", argv[i]);
 if (stat(argv[i], &buf) < 0)
 {
 printf("Error: Cannot access file %s\n", argv[i]);
 continue;
 }
 if (S_ISREG(buf.st_mode))
{
 printf("This is a regular file.\n");
 inode = buf.st_ino;
 printf("Inode of the file %s is: %ld\n", argv[i], (long)inode);
 }
 else
 {
 printf("This is not a regular file.\n");
 }
 printf("\n");
 }
 return 0;
}
Q2.#include <stdio.h>
#include <fcntl.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/wait.h>
int main()
{
 int pid;
 char file[30];
 pid = fork();
 if (pid < 0)
 {
 printf("Fork failed\n");
 exit(-1);
 }
 else if (pid == 0)
 {
 printf("Enter the file name: ");
 scanf("%s", file);
 // Execute ls -l <file>
 execlp("/bin/ls", "ls", "-l", file, NULL);
 printf("Exec failed\n"); // runs only if exec fails
 exit(1);
 }
 else
 {
 wait(NULL);
 printf("Child completed\n");
 exit(0);
 }
 return 0;
}
