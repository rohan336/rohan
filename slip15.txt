Q1.#include <stdio.h>
#include <dirent.h>
#include <sys/stat.h>
int main()
{
 DIR *dp;
 struct dirent *ep;
 struct stat sb;
 int n;
 printf("Enter size in bytes: ");
 scanf("%d", &n);
 dp = opendir("."); // open current directory
 if (dp != NULL)
 {
 while ((ep = readdir(dp)) != NULL)
 {
 // get file info using stat()
 if (stat(ep->d_name, &sb) == 0)
 {
 // check if it's a file and size > n
 if (S_ISREG(sb.st_mode) && sb.st_size > n)
 {
 printf("%s --> %ld bytes\n", ep->d_name, sb.st_size);
}
 }
 }
 closedir(dp);
 }
 else
 {
 printf("Cannot open directory!\n");
 }
 return 0;
}
Q2.#include <signal.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/wait.h>
#include <unistd.h>
pid_t child_pid;
// Handler for SIGCHLD
void sigchld_handler(int sig)
{
 printf("Received SIGCHLD signal\n");
}
// Handler for SIGALRM
void sigalrm_handler(int sig)
{
 printf("Received SIGALRM signal. Killing child process...\n");
 kill(child_pid, SIGKILL);
}
int main()
{
 // Set signal handlers
 signal(SIGCHLD, sigchld_handler);
 signal(SIGALRM, sigalrm_handler
child_pid = fork();
 if (child_pid < 0)
 {
 perror("fork");
 exit(1);
 }
 else if (child_pid == 0)
 {
 // Child process runs a command
 execlp("/bin/ls", "ls", NULL);
 perror("exec");
 exit(1);
 }
 else
 {
 rm(5); // Child must finish within 5 sec
 int status;
 wait(&status);
 alarm(0); // Cancel alarm when child is done
 if (WIFEXITED(status))
 {
 printf("Child exited normally with status %d\n",
 WEXITSTATUS(status));
 }
 else if (WIFSIGNALED(status))
 {
 printf("Child was killed by signal %d\n",
 WTERMSIG(status));
 }
 }
 return 0;
}
